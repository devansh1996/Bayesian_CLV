"""
src/data.py
Data loading, cleaning, temporal splitting, and customer-level aggregation
for the UCI Online Retail II dataset.
"""
 
import pandas as pd
import numpy as np
from pathlib import Path
from typing import Tuple
 
# ──────────────────────────────────────────────
# CONSTANTS
# ──────────────────────────────────────────────
RAW_PATH = Path("data/raw/online_retail_II.xlsx")
PROCESSED_DIR = Path("data/processed")
CAL_END = "2011-03-01"
 
# StockCodes that represent operational charges, not product purchases
NON_PRODUCT_CODES = {
    "POST", "DOT", "M", "AMAZONFEE", "BANK CHARGES",
    "C2", "D", "CRUK", "S", "PADS",
}
 
# Minimum number of countries to keep as named segments;
# rest collapsed into "Other"
MIN_COUNTRY_CUSTOMERS = 30
 
 
# ──────────────────────────────────────────────
# LOADING
# ──────────────────────────────────────────────
def load_raw(path: Path = RAW_PATH) -> pd.DataFrame:
    """Load both sheets of the UCI Online Retail II xlsx and concatenate."""
    df1 = pd.read_excel(path, sheet_name="Year 2009-2010")
    df2 = pd.read_excel(path, sheet_name="Year 2010-2011")
    df = pd.concat([df1, df2], ignore_index=True)
    df.columns = df.columns.str.strip()
    return df
 
 
# ──────────────────────────────────────────────
# CLEANING
# ──────────────────────────────────────────────
def clean_transactions(df: pd.DataFrame) -> pd.DataFrame:
    """
    Apply all cleaning steps:
    1. Drop rows with missing Customer ID
    2. Remove cancellations (InvoiceNo starts with 'C')
    3. Remove non-product stock codes
    4. Remove zero/negative prices and quantities
    5. Compute line-level revenue
    """
    n_raw = len(df)
 
    # 1. Missing Customer ID
    df = df.dropna(subset=["Customer ID"])
    df["Customer ID"] = df["Customer ID"].astype(int)
 
    # 2. Cancellations
    df["Invoice"] = df["Invoice"].astype(str)
    df = df[~df["Invoice"].str.startswith("C")]
 
    # 3. Non-product codes
    df["StockCode"] = df["StockCode"].astype(str).str.strip().str.upper()
    df = df[~df["StockCode"].isin(NON_PRODUCT_CODES)]
 
    # 4. Positive quantities and prices
    df = df[(df["Quantity"] > 0) & (df["Price"] > 0)]
 
    # 5. Revenue
    df["Revenue"] = df["Quantity"] * df["Price"]
 
    # Parse dates
    df["InvoiceDate"] = pd.to_datetime(df["InvoiceDate"])
 
    print(f"Cleaning: {n_raw:,} -> {len(df):,} rows "
          f"({n_raw - len(df):,} removed, {(n_raw-len(df))/n_raw:.1%})")
 
    return df.reset_index(drop=True)
 
 
# ──────────────────────────────────────────────
# TEMPORAL SPLIT
# ──────────────────────────────────────────────
def temporal_split(
    df: pd.DataFrame,
    cal_end: str = CAL_END,
) -> Tuple[pd.DataFrame, pd.DataFrame]:
    """
    Split transaction-level data into calibration and holdout periods.
 
    Parameters
    ----------
    df : cleaned transaction DataFrame
    cal_end : first date of holdout period (exclusive for calibration)
 
    Returns
    -------
    cal, holdout : DataFrames
    """
    cal_end_dt = pd.Timestamp(cal_end)
    cal = df[df["InvoiceDate"] < cal_end_dt].copy()
    holdout = df[df["InvoiceDate"] >= cal_end_dt].copy()
 
    print(f"Calibration: {cal['InvoiceDate'].min().date()} to "
          f"{cal['InvoiceDate'].max().date()} ({len(cal):,} rows)")
    print(f"Holdout:     {holdout['InvoiceDate'].min().date()} to "
          f"{holdout['InvoiceDate'].max().date()} ({len(holdout):,} rows)")
 
    return cal, holdout
 
 
# ──────────────────────────────────────────────
# CUSTOMER-LEVEL AGGREGATION
# ──────────────────────────────────────────────
def aggregate_customers(
    cal: pd.DataFrame,
    time_unit: str = "W",
) -> pd.DataFrame:
    """
    Aggregate calibration transactions to customer-level summary statistics
    required by BG/NBD and Gamma-Gamma models.
 
    Returns DataFrame with columns:
        customer_id, frequency, recency, T, monetary_value, country
    """
    cal_end = cal["InvoiceDate"].max() + pd.Timedelta(days=1)
 
    # Invoice-level aggregation (one row per purchase occasion)
    invoices = (
        cal
        .groupby(["Customer ID", "Invoice"])
        .agg(
            invoice_date=("InvoiceDate", "min"),
            invoice_revenue=("Revenue", "sum"),
            country=("Country", "first"),
        )
        .reset_index()
    )
 
    # Customer-level aggregation
    customers = []
    for cid, grp in invoices.groupby("Customer ID"):
        grp = grp.sort_values("invoice_date")
        first_purchase = grp["invoice_date"].min()
        last_purchase = grp["invoice_date"].max()
        n_purchases = len(grp)
 
        # Frequency = repeat purchases (total - 1)
        frequency = n_purchases - 1
 
        # Recency = time from first purchase to last purchase (in weeks)
        recency = (last_purchase - first_purchase) / np.timedelta64(1, time_unit)
 
        # T = time from first purchase to end of calibration (in weeks)
        T = (cal_end - first_purchase) / np.timedelta64(1, time_unit)
 
        # Monetary value = avg revenue per REPEAT transaction
        if frequency > 0:
            # Exclude first transaction for monetary calculation
            repeat_revenues = grp.iloc[1:]["invoice_revenue"]
            monetary_value = repeat_revenues.mean()
        else:
            monetary_value = 0.0
 
        customers.append({
            "customer_id": cid,
            "frequency": frequency,
            "recency": recency,
            "T": T,
            "monetary_value": monetary_value,
            "country": grp["country"].mode().iloc[0],
            "n_purchases": n_purchases,
            "total_revenue": grp["invoice_revenue"].sum(),
            "first_purchase": first_purchase,
            "last_purchase": last_purchase,
        })
 
    cust_df = pd.DataFrame(customers)
 
    print(f"Aggregated {len(cust_df):,} customers")
    print(f"  Repeat purchasers (freq > 0): {(cust_df['frequency'] > 0).sum():,}")
    print(f"  Mean frequency: {cust_df['frequency'].mean():.2f}")
    print(f"  Mean monetary:  {cust_df.loc[cust_df['frequency']>0, 'monetary_value'].mean():.2f}")
 
    return cust_df
 
 
def collapse_countries(
    cust_df: pd.DataFrame,
    min_customers: int = MIN_COUNTRY_CUSTOMERS,
) -> pd.DataFrame:
    """
    Collapse countries with fewer than min_customers into 'Other'.
    Returns DataFrame with updated 'country_segment' column.
    """
    counts = cust_df["country"].value_counts()
    keep = counts[counts >= min_customers].index.tolist()
 
    cust_df = cust_df.copy()
    cust_df["country_segment"] = cust_df["country"].where(
        cust_df["country"].isin(keep), other="Other"
    )
 
    print(f"Country segments: {cust_df['country_segment'].nunique()} "
          f"(collapsed {len(counts) - len(keep)} small countries into 'Other')")
    print(cust_df["country_segment"].value_counts().to_string())
 
    return cust_df
 
 
# ──────────────────────────────────────────────
# HOLDOUT GROUND TRUTH
# ──────────────────────────────────────────────
def compute_holdout_truth(
    holdout: pd.DataFrame,
    cal_customers: pd.DataFrame,
    time_unit: str = "W",
) -> pd.DataFrame:
    """
    Compute ground-truth holdout metrics for each calibration customer.
 
    Returns DataFrame with:
        customer_id, holdout_transactions, holdout_spend, is_active
    """
    # Invoice-level holdout aggregation
    holdout_invoices = (
        holdout
        .groupby(["Customer ID", "Invoice"])
        .agg(invoice_revenue=("Revenue", "sum"))
        .reset_index()
    )
 
    holdout_agg = (
        holdout_invoices
        .groupby("Customer ID")
        .agg(
            holdout_transactions=("Invoice", "nunique"),
            holdout_spend=("invoice_revenue", "sum"),
        )
        .reset_index()
        .rename(columns={"Customer ID": "customer_id"})
    )
 
    # Merge with calibration customers (left join: customers not in holdout get 0)
    truth = cal_customers[["customer_id"]].merge(
        holdout_agg, on="customer_id", how="left"
    )
    truth["holdout_transactions"] = truth["holdout_transactions"].fillna(0).astype(int)
    truth["holdout_spend"] = truth["holdout_spend"].fillna(0.0)
    truth["is_active"] = (truth["holdout_transactions"] > 0).astype(int)
 
    print(f"Holdout truth for {len(truth):,} customers:")
    print(f"  Active in holdout: {truth['is_active'].sum():,} "
          f"({truth['is_active'].mean():.1%})")
    print(f"  Mean holdout transactions: {truth['holdout_transactions'].mean():.2f}")
    print(f"  Mean holdout spend: {truth['holdout_spend'].mean():.2f}")
 
    return truth
 
 
# ──────────────────────────────────────────────
# PIPELINE
# ──────────────────────────────────────────────
def run_pipeline():
    """Run full data preparation pipeline and save processed files."""
    PROCESSED_DIR.mkdir(parents=True, exist_ok=True)
 
    # Load and clean
    df = load_raw()
    df = clean_transactions(df)
 
    # Split
    cal, holdout = temporal_split(df)
 
    # Aggregate
    customers = aggregate_customers(cal)
    customers = collapse_countries(customers)
 
    # Holdout truth
    truth = compute_holdout_truth(holdout, customers)
 
    # Save
    cal.to_parquet(PROCESSED_DIR / "cal_transactions.parquet", index=False)
    holdout.to_parquet(PROCESSED_DIR / "holdout_transactions.parquet", index=False)
    customers.to_parquet(PROCESSED_DIR / "customers.parquet", index=False)
    truth.to_parquet(PROCESSED_DIR / "holdout_truth.parquet", index=False)
 
    print(f"\nSaved to {PROCESSED_DIR}/")
    return df, cal, holdout, customers, truth
 
 
if __name__ == "__main__":
    run_pipeline()

"""
src/models.py
Bayesian CLV models: BG/NBD, Gamma-Gamma, and hierarchical extensions.
All models built in PyMC with ArviZ InferenceData output.
"""
 
import pymc as pm
import numpy as np
import pandas as pd
import arviz as az
from typing import Optional, Dict, Any
from pathlib import Path
 
TRACES_DIR = Path("outputs/traces")
 
 
# ──────────────────────────────────────────────
# STANDARD BG/NBD
# ──────────────────────────────────────────────
def build_bgnbd(
    data: pd.DataFrame,
    priors: Optional[Dict[str, Any]] = None,
) -> pm.Model:
    """
    Build a standard (non-hierarchical) BG/NBD model.
 
    Parameters
    ----------
    data : customer-level DataFrame with columns:
           frequency, recency, T
    priors : optional dict overriding default prior params, e.g.
             {"r_sigma": 10, "alpha_sigma": 10, "a_sigma": 10, "b_sigma": 10}
 
    Returns
    -------
    PyMC Model object (not yet sampled)
    """
    p = priors or {}
    r_sigma = p.get("r_sigma", 10)
    alpha_sigma = p.get("alpha_sigma", 10)
    a_sigma = p.get("a_sigma", 10)
    b_sigma = p.get("b_sigma", 10)
 
    x = data["frequency"].values.astype(float)
    t_x = data["recency"].values.astype(float)
    T = data["T"].values.astype(float)
 
    with pm.Model() as model:
        # Population-level parameters
        r = pm.HalfNormal("r", sigma=r_sigma)
        alpha = pm.HalfNormal("alpha", sigma=alpha_sigma)
        a = pm.HalfNormal("a", sigma=a_sigma)
        b = pm.HalfNormal("b", sigma=b_sigma)
 
        # Individual-level latent parameters (integrated out analytically)
        # BG/NBD log-likelihood (Fader, Hardie, Lee 2005 Eq. 7)
        ln_A_1 = (
            pm.math.gammaln(r + x)
            - pm.math.gammaln(r)
            + r * pm.math.log(alpha)
            - (r + x) * pm.math.log(alpha + T)
            + pm.math.gammaln(a + b)
            + pm.math.gammaln(b + x)
            - pm.math.gammaln(b)
            - pm.math.gammaln(a + b + x)
        )
 
        # Customer could have dropped out after any of their x transactions
        # A_2 term: sum over dropout after each transaction
        delta = pm.math.switch(
            pm.math.gt(x, 0),
            (
                pm.math.gammaln(r + x)
                - pm.math.gammaln(r)
                + r * pm.math.log(alpha)
                - (r + x) * pm.math.log(alpha + t_x)
                + pm.math.gammaln(a + 1 + b + x - 1)
                - pm.math.gammaln(a + b + x)
                + pm.math.log(a)
                - pm.math.log(b + x - 1)
            ),
            -np.inf,  # No dropout term for x=0 customers
        )
 
        log_likelihood = pm.math.logaddexp(ln_A_1, delta)
        pm.Potential("obs", log_likelihood.sum())
 
    return model
 
 
# ──────────────────────────────────────────────
# HIERARCHICAL BG/NBD (per-country)
# ──────────────────────────────────────────────
def build_hierarchical_bgnbd(
    data: pd.DataFrame,
    segment_col: str = "country_segment",
) -> pm.Model:
    """
    Build a hierarchical BG/NBD model with per-segment parameters.
    Uses non-centered parameterization to avoid funnel geometry.
 
    Parameters
    ----------
    data : customer-level DataFrame with frequency, recency, T,
           and segment_col column
    segment_col : column name for segment labels
 
    Returns
    -------
    PyMC Model object
    """
    segments = data[segment_col].unique()
    seg_idx = pd.Categorical(data[segment_col], categories=segments).codes
 
    x = data["frequency"].values.astype(float)
    t_x = data["recency"].values.astype(float)
    T = data["T"].values.astype(float)
 
    n_segments = len(segments)
 
    with pm.Model(coords={"segment": segments}) as model:
        # Hyperpriors
        mu_r = pm.HalfNormal("mu_r", sigma=10)
        sigma_r = pm.HalfNormal("sigma_r", sigma=5)
        mu_alpha = pm.HalfNormal("mu_alpha", sigma=10)
        sigma_alpha = pm.HalfNormal("sigma_alpha", sigma=5)
        mu_a = pm.HalfNormal("mu_a", sigma=10)
        sigma_a = pm.HalfNormal("sigma_a", sigma=5)
        mu_b = pm.HalfNormal("mu_b", sigma=10)
        sigma_b = pm.HalfNormal("sigma_b", sigma=5)
 
        # Non-centered parameterization
        z_r = pm.Normal("z_r", 0, 1, dims="segment")
        z_alpha = pm.Normal("z_alpha", 0, 1, dims="segment")
        z_a = pm.Normal("z_a", 0, 1, dims="segment")
        z_b = pm.Normal("z_b", 0, 1, dims="segment")
 
        # Segment-level parameters (softplus to ensure positivity)
        r = pm.Deterministic("r", pm.math.softplus(mu_r + sigma_r * z_r), dims="segment")
        alpha = pm.Deterministic("alpha", pm.math.softplus(mu_alpha + sigma_alpha * z_alpha), dims="segment")
        a = pm.Deterministic("a", pm.math.softplus(mu_a + sigma_a * z_a), dims="segment")
        b = pm.Deterministic("b", pm.math.softplus(mu_b + sigma_b * z_b), dims="segment")
 
        # Index into per-segment parameters for each customer
        r_i = r[seg_idx]
        alpha_i = alpha[seg_idx]
        a_i = a[seg_idx]
        b_i = b[seg_idx]
 
        # BG/NBD log-likelihood (same as standard but with per-customer params)
        ln_A_1 = (
            pm.math.gammaln(r_i + x)
            - pm.math.gammaln(r_i)
            + r_i * pm.math.log(alpha_i)
            - (r_i + x) * pm.math.log(alpha_i + T)
            + pm.math.gammaln(a_i + b_i)
            + pm.math.gammaln(b_i + x)
            - pm.math.gammaln(b_i)
            - pm.math.gammaln(a_i + b_i + x)
        )
 
        delta = pm.math.switch(
            pm.math.gt(x, 0),
            (
                pm.math.gammaln(r_i + x)
                - pm.math.gammaln(r_i)
                + r_i * pm.math.log(alpha_i)
                - (r_i + x) * pm.math.log(alpha_i + t_x)
                + pm.math.gammaln(a_i + 1 + b_i + x - 1)
                - pm.math.gammaln(a_i + b_i + x)
                + pm.math.log(a_i)
                - pm.math.log(b_i + x - 1)
            ),
            -np.inf,
        )
 
        log_likelihood = pm.math.logaddexp(ln_A_1, delta)
        pm.Potential("obs", log_likelihood.sum())
 
    return model
 
 
# ──────────────────────────────────────────────
# GAMMA-GAMMA MONETARY MODEL
# ──────────────────────────────────────────────
def build_gamma_gamma(
    data: pd.DataFrame,
    priors: Optional[Dict[str, Any]] = None,
) -> pm.Model:
    """
    Build the Gamma-Gamma model for monetary value.
    Only uses customers with frequency > 0.
 
    Parameters
    ----------
    data : customer-level DataFrame with frequency, monetary_value
 
    Returns
    -------
    PyMC Model object
    """
    p = priors or {}
    repeat = data[data["frequency"] > 0].copy()
 
    x = repeat["frequency"].values.astype(float)
    m = repeat["monetary_value"].values.astype(float)
 
    with pm.Model() as model:
        # Population parameters
        p_param = pm.HalfNormal("p", sigma=p.get("p_sigma", 10))
        q = pm.HalfNormal("q", sigma=p.get("q_sigma", 10))
        gamma = pm.HalfNormal("gamma", sigma=p.get("gamma_sigma", 10))
 
        # Gamma-Gamma log-likelihood (Fader & Hardie 2013)
        log_lik = (
            pm.math.gammaln(p_param * x + q)
            - pm.math.gammaln(p_param * x)
            - pm.math.gammaln(q)
            + q * pm.math.log(gamma)
            + (p_param * x - 1) * pm.math.log(m)
            + (p_param * x) * pm.math.log(x)
            - (p_param * x + q) * pm.math.log(x * m + gamma)
        )
 
        pm.Potential("obs", log_lik.sum())
 
    return model
 
 
# ──────────────────────────────────────────────
# PREDICTION UTILITIES
# ──────────────────────────────────────────────
def predict_conditional_transactions(
    trace: az.InferenceData,
    data: pd.DataFrame,
    t_future: float,
    n_samples: int = 1000,
) -> np.ndarray:
    """
    Compute E[X(t) | x, t_x, T] for each customer using posterior samples.
 
    Parameters
    ----------
    trace : ArviZ InferenceData from fitted BG/NBD
    data : customer-level DataFrame
    t_future : future period length in same time units as T
    n_samples : number of posterior samples to use
 
    Returns
    -------
    Array of shape (n_samples, n_customers) with predicted transaction counts
    """
    posterior = trace.posterior
    r = posterior["r"].values.flatten()[:n_samples]
    alpha = posterior["alpha"].values.flatten()[:n_samples]
    a = posterior["a"].values.flatten()[:n_samples]
    b = posterior["b"].values.flatten()[:n_samples]
 
    x = data["frequency"].values
    t_x = data["recency"].values
    T = data["T"].values
 
    predictions = np.zeros((n_samples, len(data)))
 
    for s in range(n_samples):
        # P(alive) for each customer
        p_alive = _compute_p_alive(r[s], alpha[s], a[s], b[s], x, t_x, T)
 
        # Expected transactions if alive
        expected_if_alive = (
            (r[s] + x) / (alpha[s] + T) * t_future
        )
 
        predictions[s] = p_alive * expected_if_alive
 
    return predictions
 
 
def _compute_p_alive(r, alpha, a, b, x, t_x, T):
    """
    Compute P(alive at T) for each customer given BG/NBD parameters.
    Fader, Hardie, Lee (2005) Eq. 11.
    """
    from scipy.special import gammaln, betaln
 
    # Numerator: customer is still alive
    log_num = -(r + x) * np.log(alpha + T) + betaln(a, b + x)
 
    # Denominator includes dropout possibility
    log_denom_alive = log_num.copy()
    log_denom_dead = np.full_like(x, -np.inf, dtype=float)
 
    mask = x > 0
    log_denom_dead[mask] = (
        -(r + x[mask]) * np.log(alpha + t_x[mask])
        + np.log(a) - np.log(b + x[mask] - 1)
        + betaln(a + 1, b + x[mask] - 1)
    )
 
    log_denom = np.logaddexp(log_denom_alive, log_denom_dead)
    p_alive = np.exp(log_num - log_denom)
 
    return np.clip(p_alive, 0, 1)
 
 
def predict_monetary_value(
    trace: az.InferenceData,
    data: pd.DataFrame,
    n_samples: int = 1000,
) -> np.ndarray:
    """
    Predict expected average transaction value using Gamma-Gamma posterior.
 
    Returns
    -------
    Array of shape (n_samples, n_customers) with predicted monetary values
    """
    posterior = trace.posterior
    p_param = posterior["p"].values.flatten()[:n_samples]
    q = posterior["q"].values.flatten()[:n_samples]
    gamma = posterior["gamma"].values.flatten()[:n_samples]
 
    repeat = data[data["frequency"] > 0]
    x = repeat["frequency"].values
    m = repeat["monetary_value"].values
 
    predictions = np.zeros((n_samples, len(repeat)))
 
    for s in range(n_samples):
        # E[M | x, m, p, q, gamma] = (q * gamma + x * p * m) / (q + x * p - 1)
        # Only valid when q + x * p > 1
        denom = q[s] + x * p_param[s] - 1
        denom = np.maximum(denom, 1e-6)  # numerical safety
        predictions[s] = (q[s] * gamma[s] + x * p_param[s] * m) / denom
 
    return predictions
 
 
def compute_clv_posterior(
    tx_predictions: np.ndarray,
    monetary_predictions: np.ndarray,
    margin: float = 1.0,
    discount_rate: float = 0.0,
) -> np.ndarray:
    """
    Compute CLV posterior samples = E[transactions] * E[monetary] * margin.
 
    Parameters
    ----------
    tx_predictions : (n_samples, n_customers) predicted transactions
    monetary_predictions : (n_samples, n_customers) predicted monetary value
    margin : profit margin fraction (default 1.0 = revenue-based CLV)
    discount_rate : periodic discount rate (default 0.0 = no discounting)
 
    Returns
    -------
    Array of shape (n_samples, n_customers) with CLV posterior samples
    """
    return tx_predictions * monetary_predictions * margin

def fit_model(
    model: pm.Model,
    draws: int = 1000,
    tune: int = 1000,
    chains: int = 4,
    target_accept: float = 0.9,
    random_seed: int = 42,
    save_name: Optional[str] = None,
) -> az.InferenceData:
    """
    Sample from a PyMC model with NUTS and return ArviZ InferenceData.
    Optionally saves trace to netCDF.
    """
    with model:
        trace = pm.sample(
            draws=draws,
            tune=tune,
            chains=chains,
            target_accept=target_accept,
            random_seed=random_seed,
            return_inferencedata=True,
        )
 
    if save_name:
        TRACES_DIR.mkdir(parents=True, exist_ok=True)
        out_path = TRACES_DIR / f"{save_name}.nc"
        trace.to_netcdf(str(out_path))
        print(f"Trace saved to {out_path}")
 
    return trace
 
 
def load_trace(name: str) -> az.InferenceData:
    """Load a previously saved ArviZ InferenceData from netCDF."""
    path = TRACES_DIR / f"{name}.nc"
    return az.from_netcdf(str(path))

"""
src/baselines.py
Baseline CLV models: RFM heuristic and XGBoost regression.
"""
 
import pandas as pd
import numpy as np
from sklearn.model_selection import TimeSeriesSplit
from sklearn.metrics import mean_absolute_error, mean_squared_error
import xgboost as xgb
 
 
# ──────────────────────────────────────────────
# RFM HEURISTIC BASELINE
# ──────────────────────────────────────────────
def rfm_heuristic_predict(
    customers: pd.DataFrame,
    holdout_length_weeks: float,
) -> pd.DataFrame:
    """
    Predict holdout spend using simple RFM heuristic:
    predicted_spend = (frequency / T) * holdout_length * avg_monetary
 
    This represents a naive extrapolation of historical purchase rate
    and average spend into the future.
    """
    df = customers.copy()
 
    # Purchase rate per week
    df["purchase_rate"] = df["n_purchases"] / df["T"].clip(lower=1e-6)
 
    # Predicted transactions in holdout
    df["pred_transactions"] = df["purchase_rate"] * holdout_length_weeks
 
    # Predicted spend (use avg transaction value for customers with repeats)
    avg_spend = df["total_revenue"] / df["n_purchases"]
    df["pred_spend"] = df["pred_transactions"] * avg_spend
 
    # Apply recency weighting: discount predictions for inactive customers
    max_recency = df["T"].max()
    recency_ratio = 1 - (df["T"] - df["recency"]) / max_recency
    df["pred_spend_weighted"] = df["pred_spend"] * recency_ratio.clip(lower=0)
 
    return df[["customer_id", "pred_transactions", "pred_spend", "pred_spend_weighted"]]
 
 
# ──────────────────────────────────────────────
# XGBOOST REGRESSION BASELINE
# ──────────────────────────────────────────────
def engineer_features(
    customers: pd.DataFrame,
    cal_transactions: pd.DataFrame,
) -> pd.DataFrame:
    """
    Engineer features from calibration data for XGBoost model.
    """
    df = customers.copy()
 
    # Core RFM features (already in customer df)
    features = df[["customer_id", "frequency", "recency", "T",
                    "monetary_value", "n_purchases", "total_revenue"]].copy()
 
    # Derived ratios
    features["purchase_rate"] = features["n_purchases"] / features["T"].clip(lower=1e-6)
    features["avg_spend"] = features["total_revenue"] / features["n_purchases"]
    features["recency_ratio"] = features["recency"] / features["T"].clip(lower=1e-6)
 
    # Time since last purchase (inactivity)
    features["inactivity"] = features["T"] - features["recency"]
 
    # Inter-purchase time stats
    ipt_stats = _compute_ipt_features(cal_transactions)
    features = features.merge(ipt_stats, on="customer_id", how="left")
 
    # Temporal features from transaction dates
    temporal = _compute_temporal_features(cal_transactions)
    features = features.merge(temporal, on="customer_id", how="left")
 
    # Country one-hot (top countries only)
    country_dummies = pd.get_dummies(
        df["country_segment"], prefix="country", drop_first=True
    )
    features = pd.concat([features, country_dummies], axis=1)
 
    return features.fillna(0)
 
 
def _compute_ipt_features(cal: pd.DataFrame) -> pd.DataFrame:
    """Compute inter-purchase time statistics per customer."""
    results = []
    for cid, grp in cal.groupby("Customer ID"):
        dates = grp["InvoiceDate"].sort_values().unique()
        if len(dates) < 2:
            results.append({"customer_id": cid, "ipt_mean": 0, "ipt_std": 0})
            continue
        diffs = np.diff(dates).astype("timedelta64[D]").astype(float)
        results.append({
            "customer_id": cid,
            "ipt_mean": diffs.mean(),
            "ipt_std": diffs.std() if len(diffs) > 1 else 0,
        })
    return pd.DataFrame(results)
 
 
def _compute_temporal_features(cal: pd.DataFrame) -> pd.DataFrame:
    """Compute day-of-week and month purchase distributions."""
    cal = cal.copy()
    cal["dow"] = cal["InvoiceDate"].dt.dayofweek
    cal["month"] = cal["InvoiceDate"].dt.month
 
    # Fraction of purchases on each day of week
    dow_dist = (
        cal.groupby(["Customer ID", "dow"]).size()
        .unstack(fill_value=0)
        .div(cal.groupby("Customer ID").size(), axis=0)
    )
    dow_dist.columns = [f"dow_frac_{d}" for d in dow_dist.columns]
 
    # Spend trend: last-quarter avg vs first-quarter avg
    cal = cal.sort_values("InvoiceDate")
    trends = []
    for cid, grp in cal.groupby("Customer ID"):
        n = len(grp)
        if n < 4:
            trends.append({"Customer ID": cid, "spend_trend": 0})
            continue
        q = n // 4
        early = grp.iloc[:q]["Revenue"].mean()
        late = grp.iloc[-q:]["Revenue"].mean()
        trend = (late - early) / (early + 1e-6)
        trends.append({"Customer ID": cid, "spend_trend": trend})
 
    trend_df = pd.DataFrame(trends).set_index("Customer ID")
 
    features = dow_dist.join(trend_df, how="outer").reset_index()
    features = features.rename(columns={"Customer ID": "customer_id"})
    return features
 
 
def train_xgboost(
    features: pd.DataFrame,
    target: pd.Series,
    n_splits: int = 5,
) -> tuple:
    """
    Train XGBoost with cross-validated hyperparameter selection.
 
    Returns
    -------
    (model, cv_scores, feature_importances)
    """
    X = features.drop(columns=["customer_id"])
    y = target.values
 
    # Hyperparameter grid (simplified for thesis)
    params = {
        "objective": "reg:squarederror",
        "max_depth": 5,
        "learning_rate": 0.05,
        "n_estimators": 500,
        "subsample": 0.8,
        "colsample_bytree": 0.8,
        "min_child_weight": 5,
        "reg_alpha": 0.1,
        "reg_lambda": 1.0,
        "random_state": 42,
        "early_stopping_rounds": 50,
    }
 
    # Time-series aware cross-validation
    tscv = TimeSeriesSplit(n_splits=n_splits)
    cv_scores = []
 
    for train_idx, val_idx in tscv.split(X):
        X_train, X_val = X.iloc[train_idx], X.iloc[val_idx]
        y_train, y_val = y[train_idx], y[val_idx]
 
        model = xgb.XGBRegressor(**params)
        model.fit(
            X_train, y_train,
            eval_set=[(X_val, y_val)],
            verbose=False,
        )
        preds = model.predict(X_val)
        cv_scores.append({
            "mae": mean_absolute_error(y_val, preds),
            "rmse": np.sqrt(mean_squared_error(y_val, preds)),
        })
 
    # Final model on full data
    final_model = xgb.XGBRegressor(**{k: v for k, v in params.items()
                                       if k != "early_stopping_rounds"})
    final_model.set_params(n_estimators=300)
    final_model.fit(X, y)
 
    importances = pd.Series(
        final_model.feature_importances_, index=X.columns
    ).sort_values(ascending=False)
 
    return final_model, pd.DataFrame(cv_scores), importances

"""
src/evaluation.py
Evaluation metrics, calibration analysis, and targeting simulation.
"""
 
import pandas as pd
import numpy as np
from sklearn.metrics import (
    mean_absolute_error, mean_squared_error,
    roc_auc_score, brier_score_loss,
)
from typing import Dict, Optional
 
 
# ──────────────────────────────────────────────
# POINT PREDICTION METRICS
# ──────────────────────────────────────────────
def compute_metrics(
    actual: np.ndarray,
    predicted: np.ndarray,
    label: str = "",
) -> Dict[str, float]:
    """Compute MAE, RMSE, MAPE for a set of predictions."""
    mae = mean_absolute_error(actual, predicted)
    rmse = np.sqrt(mean_squared_error(actual, predicted))
    # MAPE (avoid division by zero)
    nonzero = actual > 0
    mape = np.mean(np.abs(actual[nonzero] - predicted[nonzero]) / actual[nonzero]) * 100
 
    metrics = {"model": label, "MAE": mae, "RMSE": rmse, "MAPE": mape}
    return metrics
 
 
def model_comparison_table(
    actual: np.ndarray,
    predictions: Dict[str, np.ndarray],
) -> pd.DataFrame:
    """
    Build a comparison table across multiple models.
 
    Parameters
    ----------
    actual : ground truth values
    predictions : dict of {model_name: predicted_values}
    """
    rows = []
    for name, preds in predictions.items():
        rows.append(compute_metrics(actual, preds, label=name))
    return pd.DataFrame(rows).set_index("model")
 
 
# ──────────────────────────────────────────────
# CALIBRATION / COVERAGE ANALYSIS
# ──────────────────────────────────────────────
def coverage_analysis(
    actual: np.ndarray,
    posterior_samples: np.ndarray,
    levels: list = [0.50, 0.80, 0.90, 0.95],
) -> pd.DataFrame:
    """
    Compute credible interval coverage for Bayesian predictions.
 
    Parameters
    ----------
    actual : (n_customers,) ground truth
    posterior_samples : (n_samples, n_customers) posterior predictive draws
    levels : credible interval levels to evaluate
 
    Returns
    -------
    DataFrame with nominal_level, lower, upper, observed_coverage, avg_width
    """
    rows = []
    for level in levels:
        alpha = (1 - level) / 2
        lower = np.quantile(posterior_samples, alpha, axis=0)
        upper = np.quantile(posterior_samples, 1 - alpha, axis=0)
 
        covered = ((actual >= lower) & (actual <= upper)).mean()
        avg_width = (upper - lower).mean()
 
        rows.append({
            "nominal_level": f"{level:.0%}",
            "observed_coverage": f"{covered:.1%}",
            "avg_width": f"{avg_width:.2f}",
            "coverage_ratio": covered / level,
        })
    return pd.DataFrame(rows)
 
 
def calibration_by_decile(
    actual: np.ndarray,
    predicted: np.ndarray,
    n_bins: int = 10,
) -> pd.DataFrame:
    """
    Bin customers by predicted value and compute mean actual vs predicted
    per bin for calibration plots.
    """
    df = pd.DataFrame({"actual": actual, "predicted": predicted})
    df["decile"] = pd.qcut(df["predicted"], n_bins, labels=False, duplicates="drop")
 
    cal = df.groupby("decile").agg(
        mean_predicted=("predicted", "mean"),
        mean_actual=("actual", "mean"),
        count=("actual", "count"),
    ).reset_index()
 
    return cal
 
 
# ──────────────────────────────────────────────
# P(ALIVE) EVALUATION
# ──────────────────────────────────────────────
def evaluate_p_alive(
    p_alive: np.ndarray,
    is_active: np.ndarray,
) -> Dict[str, float]:
    """
    Evaluate P(alive) predictions against binary holdout activity.
 
    Returns dict with AUC, Brier score, and decile reliability data.
    """
    auc = roc_auc_score(is_active, p_alive)
    brier = brier_score_loss(is_active, p_alive)
 
    return {
        "AUC": auc,
        "Brier_score": brier,
    }
 
 
# ──────────────────────────────────────────────
# DECISION-THEORETIC TARGETING (RQ3 / H3)
# ──────────────────────────────────────────────
def targeting_simulation(
    clv_posterior: np.ndarray,
    actual_holdout_value: np.ndarray,
    cost_per_customer: float,
    targeting_depths: list = [0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50],
) -> pd.DataFrame:
    """
    Compare point-estimate vs posterior-probability targeting strategies.
 
    Parameters
    ----------
    clv_posterior : (n_samples, n_customers) CLV posterior draws
    actual_holdout_value : (n_customers,) realized holdout value
    cost_per_customer : per-customer intervention cost
    targeting_depths : fraction of customers to target
 
    Returns
    -------
    DataFrame comparing cumulative realized value for each strategy at each depth
    """
    n_customers = len(actual_holdout_value)
    clv_mean = clv_posterior.mean(axis=0)
    p_above_cost = (clv_posterior > cost_per_customer).mean(axis=0)
 
    # Net value = realized value - cost
    net_value = actual_holdout_value - cost_per_customer
 
    results = []
    for depth in targeting_depths:
        k = int(depth * n_customers)
 
        # Strategy 1: rank by posterior mean CLV
        idx_point = np.argsort(clv_mean)[::-1][:k]
        value_point = net_value[idx_point].sum()
 
        # Strategy 2: rank by P(CLV > cost)
        idx_prob = np.argsort(p_above_cost)[::-1][:k]
        value_prob = net_value[idx_prob].sum()
 
        # Oracle: rank by actual holdout value
        idx_oracle = np.argsort(actual_holdout_value)[::-1][:k]
        value_oracle = net_value[idx_oracle].sum()
 
        results.append({
            "targeting_depth": f"{depth:.0%}",
            "n_targeted": k,
            "point_estimate_value": value_point,
            "posterior_prob_value": value_prob,
            "oracle_value": value_oracle,
            "improvement": value_prob - value_point,
            "improvement_pct": (value_prob - value_point) / (abs(value_point) + 1e-6) * 100,
        })
 
    return pd.DataFrame(results)
 
 
# ──────────────────────────────────────────────
# COUNTRY-LEVEL EVALUATION (RQ2 / H2)
# ──────────────────────────────────────────────
def country_level_metrics(
    actual: np.ndarray,
    predictions: Dict[str, np.ndarray],
    countries: np.ndarray,
) -> pd.DataFrame:
    """
    Compute per-country MAE for each model variant.
 
    Parameters
    ----------
    actual : ground truth values
    predictions : dict of {model_name: predicted_values}
    countries : country labels per customer
 
    Returns
    -------
    Pivot table: rows = countries, columns = models, values = MAE
    """
    rows = []
    for country in np.unique(countries):
        mask = countries == country
        n = mask.sum()
        row = {"country": country, "n_customers": n}
        for name, preds in predictions.items():
            row[f"MAE_{name}"] = mean_absolute_error(actual[mask], preds[mask])
        rows.append(row)
 
    return pd.DataFrame(rows).sort_values("n_customers", ascending=False)

               linewidth=1.5, label=f"Global mean: {global_mean:.2f}")

    ax.set_yticklabels(segments)
    ax.set_xlabel(parameter)

    ax.legend()
    save_fig(fig, f"shrinkage_{parameter}")
    return fig
 
 
def plot_country_mae_comparison(country_metrics: pd.DataFrame):
    """Grouped bar chart of MAE by country for each model variant."""
    models = [c for c in country_metrics.columns if c.startswith("MAE_")]
    model_names = [c.replace("MAE_", "") for c in models]
 
    fig, ax = plt.subplots(figsize=(12, 6))
    x = np.arange(len(country_metrics))
    width = 0.8 / len(models)
    color_list = [COLORS["rfm"], COLORS["bayesian"], COLORS["hierarchical"]]
 
    for i, (col, name) in enumerate(zip(models, model_names)):
        offset = (i - len(models)/2 + 0.5) * width
        ax.bar(x + offset, country_metrics[col], width,
               label=name, color=color_list[i % len(color_list)],
               edgecolor="white")
 
    ax.set_xticks(x)
    ax.set_xticklabels(country_metrics["country"], rotation=45, ha="right")
    ax.set_ylabel("MAE")
    ax.set_title("Prediction Error by Country Segment")
    ax.legend(bbox_to_anchor=(1.02, 1), loc="upper left")
    save_fig(fig, "country_mae_comparison")
    return fig
